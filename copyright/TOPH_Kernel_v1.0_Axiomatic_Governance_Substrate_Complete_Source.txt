  return DOMAINS[axiom.domain] || DOMAINS.D0;

function getIsaBadges(isa) {
  return isa.map(flag => ({
    label: flag,
    color: flag === "SYSTEM_HALT" ? "#ff1a1a" :
           flag === "DUAL"        ? "#ff6b2b" :
           flag === "BRANCH"      ? "#e8a832" :
           flag === "PRIV"        ? "#1e6aff" : "#888"
  }));
}
